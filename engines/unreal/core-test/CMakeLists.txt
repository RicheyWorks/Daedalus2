cmake_minimum_required(VERSION 3.20)
project(DaedalusGamePorts LANGUAGES CXX)
enable_testing()
option(GAME_PORT_SANITIZERS "Check portable game cores with ASan and UBSan" OFF)

function(add_port_test name source)
    add_executable(${name} ${source})
    target_compile_features(${name} PRIVATE cxx_std_17)
    target_include_directories(${name} PRIVATE ../DaedalusLink/Source/DaedalusLink/Public)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wshadow -Werror)
        if(GAME_PORT_SANITIZERS)
            target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${name} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()

add_port_test(minesweeper_core_test minesweeper_core_test.cpp)
add_test(NAME minesweeper_source_parity COMMAND minesweeper_core_test ${CMAKE_CURRENT_SOURCE_DIR}/minesweeper_parity.txt)

add_port_test(pet_care_core_test pet_care_core_test.cpp)
add_test(NAME pet_care_source_parity COMMAND pet_care_core_test ${CMAKE_CURRENT_SOURCE_DIR}/pet_care_parity.txt)
