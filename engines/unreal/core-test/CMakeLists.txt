cmake_minimum_required(VERSION 3.16)
project(DaedalusSimulationTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
foreach(test simulation_wire simulation_core)
    add_executable(${test}_test ${test}_test.cpp)
    target_compile_features(${test}_test PRIVATE cxx_std_17)
    target_include_directories(${test}_test PRIVATE ../DaedalusLink/Source/DaedalusLink/Public)
    target_compile_definitions(${test}_test PRIVATE SIMULATION_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures/simulation-v1")
    target_link_libraries(${test}_test PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${test}_test PRIVATE /W4 /WX /EHsc)
    else()
        target_compile_options(${test}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
