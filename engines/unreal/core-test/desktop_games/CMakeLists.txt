cmake_minimum_required(VERSION 3.20)
project(DaedalusDesktopGames LANGUAGES CXX)
enable_testing()
option(DESKTOP_SANITIZERS "Use ASan and UBSan for portable desktop ports" OFF)
function(desktop_test name source fixture)
  add_executable(${name} ${source})
  target_include_directories(${name} PRIVATE ../../DaedalusLink/Source/DaedalusLink/Public)
  target_compile_features(${name} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wshadow -Werror)
    if(DESKTOP_SANITIZERS)
      target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${name} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME ${name} COMMAND ${name} ${CMAKE_CURRENT_SOURCE_DIR}/${fixture})
endfunction()
