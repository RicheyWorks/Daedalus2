cmake_minimum_required(VERSION 3.20)
project(DaedalusJavaArcadeProtocol LANGUAGES CXX)
enable_testing()
option(JAVA_ARCADE_SANITIZERS "Use ASan and UBSan for the portable bridge policy" OFF)
add_executable(java_arcade_protocol java_arcade_protocol_test.cpp)
add_executable(sudoku_saved_games sudoku_saved_games_test.cpp)
foreach(target java_arcade_protocol sudoku_saved_games)
  target_include_directories(${target} PRIVATE ../../DaedalusLink/Source/DaedalusLink/Public)
  target_compile_features(${target} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wshadow -Werror)
    if(JAVA_ARCADE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
