cmake_minimum_required(VERSION 3.20)
project(DaedalusJavaArcadeProtocol LANGUAGES CXX)
enable_testing()
option(JAVA_ARCADE_SANITIZERS "Use ASan and UBSan for the portable bridge policy" OFF)
add_executable(java_arcade_protocol java_arcade_protocol_test.cpp)
target_include_directories(java_arcade_protocol PRIVATE ../../DaedalusLink/Source/DaedalusLink/Public)
target_compile_features(java_arcade_protocol PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(java_arcade_protocol PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(java_arcade_protocol PRIVATE -Wall -Wextra -Wshadow -Werror)
  if(JAVA_ARCADE_SANITIZERS)
    target_compile_options(java_arcade_protocol PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(java_arcade_protocol PRIVATE -fsanitize=address,undefined)
  endif()
endif()
add_test(NAME java_arcade_protocol COMMAND java_arcade_protocol)
