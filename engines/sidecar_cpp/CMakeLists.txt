# Engine-free C++17 reader for Daedalus asset sidecars (sockets, hinges, material variants) + Blender->UE axis maths.
cmake_minimum_required(VERSION 3.16)
project(daedalus_sidecar CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(daedalus_sidecar STATIC src/json.cpp src/axis.cpp src/sidecar.cpp)
target_include_directories(daedalus_sidecar PUBLIC include)
if(MSVC)
  target_compile_options(daedalus_sidecar PRIVATE /W4 /permissive-)
else()
  target_compile_options(daedalus_sidecar PRIVATE -Wall -Wextra -pedantic)
endif()
add_executable(daedalus_sidecar_test tests/sidecar_test.cpp)
target_link_libraries(daedalus_sidecar_test PRIVATE daedalus_sidecar)
enable_testing()
add_test(NAME sidecar_fixtures COMMAND daedalus_sidecar_test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
